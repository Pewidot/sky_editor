cmake_minimum_required(VERSION 3.21)
project(SkylandersLevelEditor VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Widgets OpenGL)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND AND UNIX AND NOT APPLE)
    pkg_check_modules(NATIVE_XCB QUIET IMPORTED_TARGET xcb xcb-shape)
endif()
add_library(editor-native-window STATIC src/nativewindowhost.cpp src/nativewindowhost.h)
target_link_libraries(editor-native-window PUBLIC Qt6::Core Qt6::Widgets)
target_include_directories(editor-native-window PUBLIC src)
if(NATIVE_XCB_FOUND)
    target_link_libraries(editor-native-window PRIVATE PkgConfig::NATIVE_XCB)
    target_compile_definitions(editor-native-window PRIVATE SKYLANDERS_X11_HOST)
endif()

add_executable(skylanders-level-editor
    src/main.cpp src/backend.cpp src/backend.h src/window.cpp src/window.h
    src/mcp.cpp src/mcp.h src/sceneview.cpp src/sceneview.h src/gpurenderer.cpp src/gpurenderer.h src/portaldialog.cpp src/portaldialog.h
    src/nativegameview.cpp src/nativegameview.h src/previewlaunch.cpp src/previewlaunch.h)
target_link_libraries(skylanders-level-editor PRIVATE Qt6::Core Qt6::Widgets Qt6::OpenGL editor-native-window)
target_compile_definitions(skylanders-level-editor PRIVATE
    LEVEL_BUILDER_DEFAULT="${CMAKE_CURRENT_SOURCE_DIR}/../tools/level_builder.mjs")
if(MSVC)
    target_compile_options(skylanders-level-editor PRIVATE /W4)
else()
    target_compile_options(skylanders-level-editor PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
option(SKYLANDERS_GPU_TESTS "Run explicit desktop OpenGL tests (requires a working display/driver)" OFF)
if(BUILD_TESTING)
    find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
    find_program(NODE_EXECUTABLE node REQUIRED)
    add_test(NAME mcp_protocol COMMAND "${NODE_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol.mjs"
        "$<TARGET_FILE:skylanders-level-editor>"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/mock_backend.mjs")
    add_test(NAME ui_startup COMMAND skylanders-level-editor --smoke-test)
    set_tests_properties(ui_startup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(editor-gui-workflow-test tests/gui_workflow.cpp
        src/window.cpp src/backend.cpp src/sceneview.cpp src/gpurenderer.cpp src/portaldialog.cpp src/nativegameview.cpp src/previewlaunch.cpp)
    target_include_directories(editor-gui-workflow-test PRIVATE src)
    target_link_libraries(editor-gui-workflow-test PRIVATE Qt6::Core Qt6::Widgets Qt6::OpenGL editor-native-window)
    add_test(NAME ui_workflow COMMAND editor-gui-workflow-test "${NODE_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/mock_backend.mjs")
    set_tests_properties(ui_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(editor-previewlaunch-workflow-test tests/previewlaunch_workflow.cpp
        src/previewlaunch.cpp src/backend.cpp src/window.cpp src/sceneview.cpp src/gpurenderer.cpp src/portaldialog.cpp src/nativegameview.cpp)
    target_include_directories(editor-previewlaunch-workflow-test PRIVATE src)
    target_link_libraries(editor-previewlaunch-workflow-test PRIVATE Qt6::Core Qt6::Widgets Qt6::OpenGL Qt6::Test editor-native-window)
    add_test(NAME previewlaunch_ui_workflow COMMAND editor-previewlaunch-workflow-test "${NODE_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/mock_backend.mjs")
    set_tests_properties(previewlaunch_ui_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
    add_executable(editor-portal-workflow-test tests/portal_workflow.cpp src/portaldialog.cpp src/nativegameview.cpp)
    target_include_directories(editor-portal-workflow-test PRIVATE src)
    target_link_libraries(editor-portal-workflow-test PRIVATE Qt6::Core Qt6::Widgets editor-native-window)
    add_test(NAME portal_ui_workflow COMMAND editor-portal-workflow-test "${NODE_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/mock_portal.mjs")
    set_tests_properties(portal_ui_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(editor-nativegameview-workflow-test tests/nativegameview_workflow.cpp src/nativegameview.cpp)
    target_include_directories(editor-nativegameview-workflow-test PRIVATE src)
    target_link_libraries(editor-nativegameview-workflow-test PRIVATE Qt6::Core Qt6::Widgets editor-native-window)
    add_test(NAME nativegameview_ui_workflow COMMAND editor-nativegameview-workflow-test)
    set_tests_properties(nativegameview_ui_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
    add_executable(editor-nativecamera-workflow-test tests/nativecamera_workflow.cpp src/nativegameview.cpp)
    target_include_directories(editor-nativecamera-workflow-test PRIVATE src)
    target_link_libraries(editor-nativecamera-workflow-test PRIVATE Qt6::Core Qt6::Widgets editor-native-window)
    add_test(NAME nativecamera_ui_workflow COMMAND editor-nativecamera-workflow-test)
    set_tests_properties(nativecamera_ui_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
    add_executable(editor-nativecamera-live EXCLUDE_FROM_ALL tests/nativecamera_live.cpp src/portaldialog.cpp src/nativegameview.cpp)
    target_include_directories(editor-nativecamera-live PRIVATE src)
    target_link_libraries(editor-nativecamera-live PRIVATE Qt6::Core Qt6::Widgets editor-native-window)
    add_executable(editor-previewlaunch-live EXCLUDE_FROM_ALL tests/previewlaunch_live.cpp
        src/window.cpp src/backend.cpp src/sceneview.cpp src/gpurenderer.cpp src/previewlaunch.cpp src/portaldialog.cpp src/nativegameview.cpp)
    target_include_directories(editor-previewlaunch-live PRIVATE src)
    target_link_libraries(editor-previewlaunch-live PRIVATE Qt6::Core Qt6::Widgets Qt6::OpenGL editor-native-window)
    if(NATIVE_XCB_FOUND)
        add_executable(editor-nativewindow-workflow-test EXCLUDE_FROM_ALL tests/nativewindow_workflow.cpp)
        target_link_libraries(editor-nativewindow-workflow-test PRIVATE editor-native-window PkgConfig::NATIVE_XCB)
        if(SKYLANDERS_GPU_TESTS)
            set_property(TARGET editor-nativewindow-workflow-test PROPERTY EXCLUDE_FROM_ALL FALSE)
            add_test(NAME nativewindow_x11_workflow COMMAND editor-nativewindow-workflow-test)
            set_tests_properties(nativewindow_x11_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb" TIMEOUT 20)
        endif()
    endif()
    add_executable(editor-texture-render-test tests/texture_render.cpp src/sceneview.cpp src/gpurenderer.cpp)
    target_include_directories(editor-texture-render-test PRIVATE src)
    target_link_libraries(editor-texture-render-test PRIVATE Qt6::Core Qt6::Widgets Qt6::OpenGL)
    add_test(NAME texture_render COMMAND editor-texture-render-test)
    set_tests_properties(texture_render PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(SKYLANDERS_GPU_TESTS)
        add_test(NAME texture_render_gpu COMMAND editor-texture-render-test)
        set_tests_properties(texture_render_gpu PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;SKYLANDERS_RENDERER=gpu")

    add_test(NAME render_cli COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/render_cli.mjs"
        "$<TARGET_FILE:skylanders-level-editor>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/mock_backend.mjs")

install(TARGETS skylanders-level-editor RUNTIME DESTINATION bin)
